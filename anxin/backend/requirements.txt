# Runtime dependencies ONLY -- everything needed to actually run Anxin.
#
# Development tools (pytest, ruff, pyright) live in requirements-dev.txt so
# that simply running the app never requires a Rust or C toolchain. That
# matters: ruff is written in Rust, and on any Python without prebuilt wheels
# pip tries to compile it, which fails on machines with no Rust installed --
# a linter should never be able to block a demo.
#
# Version ranges rather than exact pins, deliberately: teammates and judges
# run different Python versions, and an exact pin forces a source build on any
# Python whose wheels weren't published for that specific release. Tested
# against the lower bound of each range on Python 3.11.
#
# Requires CPython 3.11-3.13 from python.org, Anaconda, or the Microsoft
# Store. NOT the MSYS2/MinGW Python bundled with some Git Bash installs --
# PyPI publishes no wheels for that platform, so every package (including
# Rust-based pydantic-core) would need compiling from source. See README.

fastapi>=0.115,<1.0
uvicorn[standard]>=0.30,<1.0
pydantic>=2.9,<3.0
pydantic-settings>=2.5,<3.0
httpx>=0.27,<1.0
python-multipart>=0.0.9,<1.0
beautifulsoup4>=4.12,<5.0

# Screenshot mode only. pytesseract is a thin wrapper -- the Tesseract binary
# itself is a separate system install (see README). The app degrades cleanly
# if it is missing; only /api/ocr stops working.
pytesseract>=0.3.13,<1.0
Pillow>=10.4
